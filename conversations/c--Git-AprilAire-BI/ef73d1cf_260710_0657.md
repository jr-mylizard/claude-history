   ```sql
   SELECT DISTINCT ON (product_id) product_id, title, title, thumbnail, category, rating, author
   FROM android_app_search_results
   WHERE product_id IS NOT NULL
   ORDER BY product_id, id DESC   -- take the most recent row per product_id
   ON CONFLICT (s_product_id) DO UPDATE SET
     s_title      = COALESCE(android_apps.s_title, EXCLUDED.s_title),
     app_name     = COALESCE(android_apps.app_name, EXCLUDED.app_name),
     s_thumbnail  = COALESCE(android_apps.s_thumbnail, EXCLUDED.s_thumbnail),
     category     = COALESCE(android_apps.category, EXCLUDED.category),
     s_rating     = COALESCE(android_apps.s_rating, EXCLUDED.s_rating),
     author_name  = COALESCE(android_apps.author_name, EXCLUDED.author_name);
   ```
   Note: `COALESCE(existing, incoming)` means we never overwrite a manually-set value — the existing `android_apps` row wins.
